{{- /* A link card. Called with the lines of a card block: the article's address on the first line. The title, description, outlet and picture come from the article itself (layouts/_partials/linkcard.html). Lines under the address can replace what was read: title: ... description: ... picture: no Some outlets do not let their pages be read (data/outlets.json, "refuses"). Their pages are not asked for. Such a card shows the outlet's name from that list, its logo where one is on file (assets/outlets), and the headline the editor typed (Austin, November 26, 2026: "create another fallback version ... the name and logo"). The card has no paragraph inside it, so its words stay out of the homepage's opening text. */ -}} {{- $lines := split (strings.TrimSpace .) "\n" -}} {{- $url := strings.TrimSpace (index $lines 0) -}} {{- $set := dict -}} {{- range after 1 $lines -}} {{- with findRESubmatch `^\s*(title|description|picture)\s*:\s*(.*?)\s*$` . 1 -}}{{- $set = merge $set (dict (index (index . 0) 1) (index (index . 0) 2)) -}}{{- end -}} {{- end -}} {{- if findRE `^https?://\S+$` $url -}} {{- $host := strings.TrimPrefix "www." (urls.Parse $url).Hostname -}} {{- $outlet := dict -}} {{- range $dom, $o := hugo.Data.outlets.outlets -}} {{- if or (eq $host $dom) (strings.HasSuffix $host (printf ".%s" $dom)) -}}{{- $outlet = $o -}}{{- end -}} {{- end -}} {{- $c := dict "ok" false "title" "" "description" "" "site" $host "image" false -}} {{- if not $outlet.refuses -}}{{- $c = partialCached "linkcard.html" $url $url -}}{{- end -}} {{- $site := $outlet.name | default $c.site -}} {{- $title := $set.title | default $c.title | default (printf "Read the article at %s" $site) -}} {{- $desc := $set.description | default $c.description -}} {{- $pic := $c.image -}}{{- if in (slice "no" "none" "off") (lower ($set.picture | default "")) -}}{{- $pic = false -}}{{- end -}} {{- $logo := false -}} {{- if not $c.ok -}}{{- with $outlet.logo -}}{{- with resources.Get . -}}{{- $logo = . -}}{{- end -}}{{- end -}}{{- end }} {{ $site }}{{ $title }}{{ with $desc }}{{ . }}{{ end }} {{ with $pic }}{{ end }} {{- if not $pic }}{{ with $logo }}{{ end }}{{ end }} {{- end -}}
