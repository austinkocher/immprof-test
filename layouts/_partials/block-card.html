{{- /* A link card. Called with the lines of a card block: the article's address on the first line. The title, description, outlet and picture come from the article itself (layouts/_partials/linkcard.html). Lines under the address can replace what was read: title: ... description: ... picture: no The card has no paragraph inside it, so its words stay out of the homepage's opening text. */ -}} {{- $lines := split (strings.TrimSpace .) "\n" -}} {{- $url := strings.TrimSpace (index $lines 0) -}} {{- $set := dict -}} {{- range after 1 $lines -}} {{- with findRESubmatch `^\s*(title|description|picture)\s*:\s*(.*?)\s*$` . 1 -}}{{- $set = merge $set (dict (index (index . 0) 1) (index (index . 0) 2)) -}}{{- end -}} {{- end -}} {{- if findRE `^https?://\S+$` $url -}} {{- $c := partialCached "linkcard.html" $url $url -}} {{- $title := $set.title | default $c.title -}} {{- $desc := $set.description | default $c.description -}} {{- $pic := $c.image -}}{{- if in (slice "no" "none" "off") (lower ($set.picture | default "")) -}}{{- $pic = false -}}{{- end -}} {{- if $title }} {{ $c.site }}{{ $title }}{{ with $desc }}{{ . }}{{ end }} {{ with $pic }}{{ end }} {{- else }}

{{ $url }}

{{- end -}} {{- end -}}
