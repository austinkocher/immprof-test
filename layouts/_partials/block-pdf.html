"\n" -}} {{- $src := strings.TrimSpace (index $lines 0) -}} {{- $title := "" -}} {{- range after 1 $lines -}}{{- with findRESubmatch `^\s*title\s*:\s*(.*?)\s*$` . 1 -}}{{- $title = index (index . 0) 1 -}}{{- end -}}{{- end -}} {{- $href := $src -}}{{- $meta := "PDF" -}} {{- with resources.Get (strings.TrimPrefix "/" $src) -}} {{- $href = .RelPermalink -}} {{- $bytes := len .Content -}} {{- $size := printf "%d KB" (div $bytes 1024) -}}{{- if ge $bytes 1048576 -}}{{- $size = printf "%.1f MB" (div (float $bytes) 1048576.0) -}}{{- end -}} {{- $pages := len (findRE `/Type\s*/Page\b` .Content) -}} {{- $meta = printf "PDF, %s" $size -}} {{- if eq $pages 1 -}}{{- $meta = printf "%s, 1 page" $meta -}}{{- else if gt $pages 1 -}}{{- $meta = printf "%s, %d pages" $meta $pages -}}{{- end -}} {{- end -}} {{- if not $title -}}{{- $title = path.BaseName $src | humanize -}}{{- end }} {{ $title }}{{ $meta }} Download
