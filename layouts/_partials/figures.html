{{- /* Captions. In a Markdown post, a line in italics directly under a picture is that picture's caption or credit (the "Picture" button on the writing page writes it that way). The picture and its line become a figure with a caption. Links are allowed inside a caption. Imported posts are left as they are. */ -}} {{- $c := string .Content -}} {{- $md := false -}}{{- with .File -}}{{- if eq .Ext "md" -}}{{- $md = true -}}{{- end -}}{{- end -}} {{- if and $md (strings.Contains $c "]*>|)*)` -}} {{- $c = replaceRE (printf `

(]*>)\s*(?:\s*)?%s

` $cap) `` $c -}} {{- $c = replaceRE (printf `

(]*>)

\s*

%s

` $cap) `` $c -}} {{- end -}} {{- return $c -}}
